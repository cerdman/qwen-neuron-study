cmake_minimum_required(VERSION 3.20)
project(neuron_study LANGUAGES C CXX)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "")
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "")
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "")
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "")
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "")
set(LLAMA_CURL OFF CACHE BOOL "")
set(GGML_NATIVE OFF CACHE BOOL "")
set(LLAMA_REVISION 6c01da6278ab6c69ed67d9124806b50d2b299ba7)
set(LLAMA_ARCHIVE "${CMAKE_CURRENT_BINARY_DIR}/llama-${LLAMA_REVISION}.tar.gz")
if(NOT FETCHCONTENT_SOURCE_DIR_LLAMA_CPP)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/fetch_source.py"
    "https://codeload.github.com/ggml-org/llama.cpp/legacy.tar.gz/${LLAMA_REVISION}" "${LLAMA_ARCHIVE}"
    COMMAND_ERROR_IS_FATAL ANY)
endif()
FetchContent_Declare(llama_cpp URL "${LLAMA_ARCHIVE}"
  URL_HASH SHA256=3862718289d8974ba509cd3efe9418aaaa95c2ffd5b684c353792f8c9fbe904a)
FetchContent_MakeAvailable(llama_cpp)
add_executable(neuron-study-engine native/engine.cpp)
target_compile_features(neuron-study-engine PRIVATE cxx_std_20)
target_link_libraries(neuron-study-engine PRIVATE llama ggml)
install(TARGETS neuron-study-engine RUNTIME DESTINATION neuron_study/bin COMPONENT study)
